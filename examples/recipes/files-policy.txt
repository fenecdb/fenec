# collection  access                  rows

# A person lists, reads and deletes their own files' metadata ...
files         read,delete             where owner = $jwt.sub

# ... and adds a file of up to 10 MiB, pending: only the backend, once the
# object is in the bucket, marks it ready.
files         insert                  where owner = $jwt.sub and state = "pending" and size > 0 and size <= 10485760

# ... and renames it.
files         update(name)            where owner = $jwt.sub

# The backend's own token. Its identity is its `sub` and an `iss` that only
# your services sign (service-token.mjs): a `sub` alone is the provider's to
# assign, and a provider's token (a Firebase project's owner picks the uid of a
# custom token) can say `files-backend`; a `role` claim would be the
# provider's to fill, and one a person may set (Supabase's user_metadata,
# Clerk's unsafe_metadata) would be theirs. The backend reads rows to check
# them, and moves a pending one to ready, a transition because one filter asked
# of the row before and after would refuse the move it grants (`new.` is the
# row as the write leaves it).
files         read                    where $jwt.sub = 'files-backend' and $jwt.iss = 'urn:fenec:services'
files         update(state)           where state = "pending" and new.state = "ready" and $jwt.sub = 'files-backend' and $jwt.iss = 'urn:fenec:services'
