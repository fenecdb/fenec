# collection  access                                       rows

# The signup function (worker/index.mjs) adds a signup and the job that welcomes
# it, and reads and changes nothing. Each service is its `sub`, which a
# provider assigns and a person does not choose (service-token.mjs).
signups       insert                                       where $jwt.sub = 'signup-function'
jobs          insert                                       where $jwt.sub = 'signup-function'

# The welcome worker (welcome-worker.mjs) claims a job, writes its lease and
# its failures, and deletes what it has done.
jobs          read,delete                                  where $jwt.sub = 'welcome-worker'
jobs          update(owner, run_at, attempts, error)       where $jwt.sub = 'welcome-worker'
