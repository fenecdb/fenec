Quickstart

Start the server, write two rows, query and search them with curl and from your language, and look at them in the studio. About five minutes. For a database in a web page with no server, see Quickstart: in the browser.

  1. Install fenec-server and start it on a file. It runs in the foreground, so the next steps go in another terminal.

    curl -fsSL https://fenecdb.com/install.sh | sh

    The script puts fenec-server and the shell in ~/.fenecdb/bin and edits no shell file: it prints the export line that puts that directory on your PATH, and the next block runs it. Add the line to ~/.zshrc or ~/.bashrc to keep it.

    What the install script prints 
    Installed fenecdb {{version}} (aarch64-apple-darwin) in /Users/you/.fenecdb/bin:
  fenec          the shell:  fenec scratch.fenec
  fenec-server   the server: fenec-server --file notes.fenec
  (fenec-server {{version}})

/Users/you/.fenecdb/bin is not on your PATH. Add it for this shell:

  export PATH="$HOME/.fenecdb/bin:$PATH"

and put the same line in your shell's startup file (~/.zshrc, ~/.bashrc
or ~/.profile) to keep it. For fish: fish_add_path "$HOME/.fenecdb/bin".

  2. Make a collection and write two rows. The language is FenecQL: a collection has typed fields, an index on each one you filter, order or search by (@text for words, @sorted for ranges and order, @hash for equality), and the values go as $1, $2 parameters, never into the text.

    curl -s http://127.0.0.1:8080/query \
  -d '{"query": "create collection notes (title text @text, year int @sorted, kind text @hash)"}'
curl -s http://127.0.0.1:8080/query \
  -d '{"query": "put notes {title: $1, year: $2, kind: $3}", "params": ["Dunes", 2021, "book"]}'
curl -s http://127.0.0.1:8080/query \
  -d '{"query": "put notes {title: $1, year: $2, kind: $3}", "params": ["Night at the oasis", 2024, "film"]}'
    {"message":"collection `notes` created"}
{"affected":1}
{"affected":1}

    curl prints each answer with no line break after it, so in a terminal they run together; they are shown one to a line here.

  3. Read the rows back by a filter and an order, by a kind, then search their words, with a typo in the search.

    curl -s http://127.0.0.1:8080/query \
  -d '{"query": "get notes where year >= $1 order year desc", "params": [2020]}'
curl -s http://127.0.0.1:8080/query \
  -d '{"query": "get notes where kind = $1", "params": ["book"]}'
curl -s http://127.0.0.1:8080/query \
  -d '{"query": "get notes select title match title $1 fuzzy", "params": ["oasys"]}'
    [{"id":2,"title":"Night at the oasis","year":2024,"kind":"film"},{"id":1,"title":"Dunes","year":2021,"kind":"book"}]
[{"id":1,"title":"Dunes","year":2021,"kind":"book"}]
[{"title":"Night at the oasis","_score":0.62239116}]

    kind is a @hash index, which answers equality and gives the studio something to count by. match searches the words of a @text field, fuzzy forgives a misspelling, and _score says how well the row matched.

  4. Do the same from your language, with its client or the HTTP client it already has. Each program writes a third row, reads the rows from 2020 on, searches a word spelled wrong and ends on a refused read, with its status and the server's message. Run it once: it writes the row every time.

  const res = await fetch('http://127.0.0.1:8080/query', {

    package main

import (
	"context"
	"errors"
	"fmt"
	"log"
	"strings"

	fenecdb "github.com/fenecdb/fenecdb/integrations/go"
)

type Note struct {
	Title string `json:"title"`
	Year  int    `json:"year"`
}

func titles(notes []Note) string {
	var out []string
	for _, n := range notes {
		out = append(out, n.Title)
	}
	return strings.Join(out, ", ")
}

func main() {
	ctx := context.Background()
	db := fenecdb.New("http://127.0.0.1:8080")

	if _, err := db.Exec(ctx, "put notes {title: $1, year: $2, kind: $3}", "Sand and stars", 2023, "book"); err != nil {
		log.Fatal(err)
	}

	rows, err := fenecdb.QueryAs[Note](ctx, db, "get notes where year >= $1 order year desc", 2020)
	if err != nil {
		log.Fatal(err)
	}
	fmt.Println("from 2020:", titles(rows))

	hits, err := fenecdb.QueryAs[Note](ctx, db, "get notes select title match title $1 fuzzy", "oasys")
	if err != nil {
		log.Fatal(err)
	}
	fmt.Println("search:", titles(hits))

	var refused *fenecdb.Error
	if _, err := db.Query(ctx, "get nowhere"); errors.As(err, &refused) {
		fmt.Println("refused:", refused.Status, refused.Message)
	}
}

    use serde_json::{json, Value};

fn query(q: &str, params: Value) -> Result<Value, ureq::Error> {
    Ok(ureq::post("http://127.0.0.1:8080/query")
        .send_json(json!({"query": q, "params": params}))?
        .into_json()?)
}

fn titles(rows: &Value) -> String {
    let titles: Vec<&str> = rows.as_array().unwrap().iter().filter_map(|r| r["title"].as_str()).collect();
    titles.join(", ")
}

fn main() -> Result<(), ureq::Error> {
    query("put notes {title: $1, year: $2, kind: $3}", json!(["Sand and stars", 2023, "book"]))?;

    let rows = query("get notes where year >= $1 order year desc", json!([2020]))?;
    println!("from 2020: {}", titles(&rows));

    let hits = query("get notes select title match title $1 fuzzy", json!(["oasys"]))?;
    println!("search: {}", titles(&hits));

    // A refusal is `Error::Status` with the server's status, its body {"error": ...}.
    match query("get nowhere", json!([])) {
        Err(ureq::Error::Status(status, res)) => {
            let body: Value = res.into_json()?;
            println!("refused: {status} {}", body["error"].as_str().unwrap_or_default());
        }
        other => {
            other?;
        }
    }
    Ok(())
}

    import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;
import java.util.ArrayList;
import java.util.regex.Pattern;

public class Notes {
    static final HttpClient HTTP = HttpClient.newHttpClient();

    // One FenecQL statement, as the JSON {"query": ..., "params": [...]}.
    static HttpResponse<String> query(String json) throws Exception {
        var req = HttpRequest.newBuilder(URI.create("http://127.0.0.1:8080/query"))
            .header("Content-Type", "application/json")
            .POST(HttpRequest.BodyPublishers.ofString(json))
            .build();
        return HTTP.send(req, HttpResponse.BodyHandlers.ofString());
    }

    // The standard library has no JSON: every "title" of an answer, or its "error".
    static String find(String key, String body) {
        var found = new ArrayList<String>();
        var m = Pattern.compile("\"" + key + "\":\"([^\"]*)\"").matcher(body);
        while (m.find()) found.add(m.group(1));
        return String.join(", ", found);
    }

    public static void main(String[] args) throws Exception {
        query("""
            {"query": "put notes {title: $1, year: $2, kind: $3}", "params": ["Sand and stars", 2023, "book"]}""");

        var rows = query("""
            {"query": "get notes where year >= $1 order year desc", "params": [2020]}""");
        System.out.println("from 2020: " + find("title", rows.body()));

        var hits = query("""
            {"query": "get notes select title match title $1 fuzzy", "params": ["oasys"]}""");
        System.out.println("search: " + find("title", hits.body()));

        var refused = query("""
            {"query": "get nowhere"}""");
        System.out.println("refused: " + refused.statusCode() + " " + find("error", refused.body()));
    }
}

    <?php

// One FenecQL statement; the answer is its status and the decoded JSON body.
function query(string $q, array $params = []): array {
    $ch = curl_init('http://127.0.0.1:8080/query');
    curl_setopt_array($ch, [
        CURLOPT_POST => true,
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_HTTPHEADER => ['Content-Type: application/json'],
        CURLOPT_POSTFIELDS => json_encode(['query' => $q, 'params' => $params]),
    ]);
    $body = json_decode(curl_exec($ch), true);
    return [curl_getinfo($ch, CURLINFO_RESPONSE_CODE), $body];
}
